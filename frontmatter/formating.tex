% --- Chapter Page Style ---
\makeatletter
\def\@makechapterhead#1{%
  {\parindent \z@ \centering \normalfont
    \interlinepenalty\@M
    \ifnum \c@secnumdepth >\m@ne
      \normalsize \MakeUppercase{\@chapapp\ \thechapter}
      \par\nobreak
      \vskip 0pt
    \fi
    \normalsize \MakeUppercase{#1}\par\nobreak
    \vskip 20pt
  }}
\def\@makeschapterhead#1{%
  {\parindent \z@ \centering \normalfont
    \interlinepenalty\@M
    \normalsize \MakeUppercase{#1}\par\nobreak
    \vskip 20pt
  }}
\makeatother
